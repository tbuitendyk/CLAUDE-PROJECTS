# YouTube download + caption retrieval
# Unlike the other deps below, this one is intentionally NOT pinned to an exact
# version: YouTube frequently changes its player/signature internals, and
# yt-dlp ships frequent point releases purely to keep extraction working.

# like "Requested format is not available" / "nsig extraction failed". Always
# install/upgrade to the latest: pip install -U yt-dlp
yt-dlp>=2024.12.13

# Linguistic rechunking: spaCy splits the transcript at real sentence/clause
# boundaries (see pipeline/rechunker.py) so chunks fall on complete thoughts
# instead of mid-sentence pauses. Pure CPU, no torch/CUDA -- deliberately
# avoided the transformer punctuation models here, which drag in multi-GB CUDA
# wheels. The rechunker degrades to punctuation-only boundaries if a model is
# ever missing.
#
# Pinned to the spaCy 3.7 line on purpose: 3.8+ requires Python >= 3.10 (its
# thinc>=8.3.12 dep has no cp39 wheel), but the target VPS runs Python 3.9, so
# 3.8 falls back to a from-source build that fails. 3.7.x ships prebuilt cp39
# wheels and supports Python 3.7-3.12. The two small language models (~13-15 MB
# each) must match that line (3.7.x).
spacy>=3.7.2,<3.8.0
https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.7.1/en_core_web_sm-3.7.1-py3-none-any.whl
https://github.com/explosion/spacy-models/releases/download/es_core_news_sm-3.7.0/es_core_news_sm-3.7.0-py3-none-any.whl

# Grammar-based punctuation restoration on run-on transcripts (a fast speaker
# whose captions/Whisper output have no punctuation defeats the pause-based
# heuristic). Runs a small ONNX model on CPU via onnxruntime + numpy -- NO
# torch/CUDA (see pipeline/punctuation_onnx.py, a torch-free port of the
# Apache-2.0 `punctuators` inference). The model (~English punct/cap/seg) is
# fetched from HuggingFace on first use / at deploy time. Entirely optional:
# the rechunker falls back to its heuristic if any of this is missing.
onnxruntime>=1.16,<2.0
sentencepiece>=0.2.0
huggingface-hub>=0.23,<1.0
pyyaml>=6.0
